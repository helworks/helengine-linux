    string(LENGTH "${${limit}}" limit_length)
    if(NOT "${${limit}}" MATCHES "^[1-9][0-9]*$" OR limit_length GREATER 10 OR "${${limit}}" GREATER 2147483647)
        message(FATAL_ERROR "${limit} must be a positive decimal Int32 integer.")
    endif()
endforeach()

    if(NOT flag_value MATCHES "^(ON|OFF|TRUE|FALSE|YES|NO|1|0)$")
        message(FATAL_ERROR "${flag} must be a CMake boolean.")
    endif()
endforeach()

foreach(stage vert frag)
    if(stage STREQUAL "vert")
        set(entry VSMain)
        set(binding_shifts)
    else()
        set(entry PSMain)
        set(binding_shifts --shift-texture-binding frag 100 --shift-sampler-binding frag 200)
    endif()
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/mesh.${stage}.spv"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
        COMMAND "${HELENGINE_GLSLANG_VALIDATOR}" -D -V --target-env vulkan1.2 --hlsl-iomap -S "${stage}" -e "${entry}" ${binding_shifts} "${CMAKE_CURRENT_SOURCE_DIR}/tests/shaders/mesh.hlsl" -o "${CMAKE_CURRENT_BINARY_DIR}/shaders/mesh.${stage}.spv"
        DEPENDS tests/shaders/mesh.hlsl VERBATIM)
    list(APPEND mesh_test_shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders/mesh.${stage}.spv")
    foreach(shader shadow_depth point_shadow_depth)
        if(shader STREQUAL "shadow_depth")
            set(alias mesh_shadow_depth)
        else()
            set(alias mesh_point_shadow)
        endif()
        add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/${alias}.${stage}.spv"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.${stage}.spv" "${CMAKE_CURRENT_BINARY_DIR}/shaders/${alias}.${stage}.spv"
            DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.${stage}.spv" VERBATIM)
        list(APPEND mesh_test_shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders/${alias}.${stage}.spv")
    endforeach()
endforeach()
