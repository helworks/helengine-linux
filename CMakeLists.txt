set(HELENGINE_CODE_ROOT "" CACHE PATH "Generated project code-module root.")
set(HELENGINE_CONTENT_ROOT "" CACHE PATH "Cooked content root owned by the build pipeline.")
set(HELENGINE_LINUX_INITIAL_WIDTH "1280" CACHE STRING "Resolved initial desktop framebuffer width.")
set(HELENGINE_LINUX_INITIAL_HEIGHT "720" CACHE STRING "Resolved initial desktop framebuffer height.")
set(HELENGINE_NATIVE_PROFILE "Debug" CACHE STRING "Native profile: Debug, Release, or Profiler.")

option(HELENGINE_LINUX_BUILD_PLAYER "Build the real generated-core player, never a diagnostic substitute." ON)

    message(FATAL_ERROR "HELENGINE_NATIVE_PROFILE must be Debug, Release, or Profiler.")
endif()
if(HELENGINE_LINUX_BUILD_PLAYER)
    if(NOT HELENGINE_LINUX_INITIAL_WIDTH MATCHES "^[1-9][0-9]*$" OR NOT HELENGINE_LINUX_INITIAL_HEIGHT MATCHES "^[1-9][0-9]*$")
        message(FATAL_ERROR "Linux initial framebuffer dimensions must be positive integers.")
    endif()
    if(NOT EXISTS "${HELENGINE_CORE_CPP_ROOT}/helcpp_config.hpp")
        message(FATAL_ERROR "HELENGINE_CORE_CPP_ROOT must contain helcpp_config.hpp. Generate the Linux core first; diagnostics are available with HELENGINE_LINUX_BUILD_PLAYER=OFF.")
    endif()

    INTERFACE_INCLUDE_DIRECTORIES "${WAYLAND_CLIENT_INCLUDE_DIR}")

    src/vulkan_context.cpp src/vulkan_renderer_2d.cpp src/vulkan_mesh_renderer.cpp src/sdl_host.cpp src/wayland_input_region.cpp)

target_compile_definitions(helengine_linux_graphics PUBLIC HELENGINE_LINUX_SHADER_ROOT="${CMAKE_CURRENT_BINARY_DIR}/shaders")
foreach(shader quad.vert quad.frag)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.spv"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
        COMMAND "${HELENGINE_GLSLANG_VALIDATOR}" -V --target-env vulkan1.1 "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}" -o "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.spv"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}" VERBATIM)
    list(APPEND compiled_shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.spv")
endforeach()
foreach(shader shadow_depth point_shadow_depth)
    foreach(stage vert frag)
        if(stage STREQUAL "vert")
            set(entry VS)
        else()
            set(entry PS)
        endif()
        add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.${stage}.spv"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
            COMMAND "${HELENGINE_GLSLANG_VALIDATOR}" -D -V --target-env vulkan1.2 --hlsl-iomap -S "${stage}" -e "${entry}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}.hlsl" -o "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.${stage}.spv"
            DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}.hlsl" VERBATIM)
        list(APPEND compiled_shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.${stage}.spv")
    endforeach()
endforeach()
add_custom_target(helengine_linux_shaders DEPENDS ${compiled_shaders})

foreach(stage vert frag)
    if(stage STREQUAL "vert")
        set(entry VSMain)
        set(binding_shifts)
    else()
        set(entry PSMain)
        set(binding_shifts --shift-texture-binding frag 100 --shift-sampler-binding frag 200)
    endif()
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/mesh.${stage}.spv"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
        COMMAND "${HELENGINE_GLSLANG_VALIDATOR}" -D -V --target-env vulkan1.2 --hlsl-iomap -S "${stage}" -e "${entry}" ${binding_shifts} "${CMAKE_CURRENT_SOURCE_DIR}/tests/shaders/mesh.hlsl" -o "${CMAKE_CURRENT_BINARY_DIR}/shaders/mesh.${stage}.spv"
        DEPENDS tests/shaders/mesh.hlsl VERBATIM)
    list(APPEND mesh_test_shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders/mesh.${stage}.spv")
    foreach(shader shadow_depth point_shadow_depth)
        if(shader STREQUAL "shadow_depth")
            set(alias mesh_shadow_depth)
        else()
            set(alias mesh_point_shadow)
        endif()
        add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/${alias}.${stage}.spv"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.${stage}.spv" "${CMAKE_CURRENT_BINARY_DIR}/shaders/${alias}.${stage}.spv"
            DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.${stage}.spv" VERBATIM)
        list(APPEND mesh_test_shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders/${alias}.${stage}.spv")
    endforeach()
endforeach()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/mesh_shadow_sample.frag.spv"
    COMMAND "${HELENGINE_GLSLANG_VALIDATOR}" -D -V --target-env vulkan1.2 --hlsl-iomap -S frag -e PSMain --shift-texture-binding frag 100 --shift-sampler-binding frag 200 "${CMAKE_CURRENT_SOURCE_DIR}/tests/shaders/shadow_sample.hlsl" -o "${CMAKE_CURRENT_BINARY_DIR}/shaders/mesh_shadow_sample.frag.spv"
    DEPENDS tests/shaders/shadow_sample.hlsl VERBATIM)
list(APPEND mesh_test_shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders/mesh_shadow_sample.frag.spv")

    endif()
    if(HELENGINE_NATIVE_PROFILE STREQUAL "Profiler")
        if(NOT EXISTS "${HELENGINE_TRACY_ROOT}/public/TracyClient.cpp")
            message(FATAL_ERROR "Profiler requires the pinned Tracy v0.13.1 checkout. Run scripts/prepare-dependencies.py or set HELENGINE_TRACY_ROOT.")
        endif()
        find_package(Git REQUIRED)
        execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${HELENGINE_TRACY_ROOT}" rev-parse HEAD OUTPUT_VARIABLE tracy_commit OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE tracy_result)
        if(NOT tracy_result EQUAL 0 OR NOT tracy_commit STREQUAL "05cceee0df3b8d7c6fa87e9638af311dbabc63cb")
            message(FATAL_ERROR "Profiler requires locked Tracy commit 05cceee0df3b8d7c6fa87e9638af311dbabc63cb.")
        endif()
        set(TRACY_ENABLE ON CACHE BOOL "Enable actual native Tracy profiling." FORCE)
        add_subdirectory("${HELENGINE_TRACY_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/tracy")
    endif()
    if(EXISTS "${HELENGINE_CORE_CPP_ROOT}/helengine_core_unity.cpp")
        set(core_unity "${HELENGINE_CORE_CPP_ROOT}/helengine_core_unity.cpp")
    elseif(EXISTS "${HELENGINE_CORE_CPP_ROOT}/generated_unity.cpp")
        set(core_unity "${HELENGINE_CORE_CPP_ROOT}/generated_unity.cpp")
    else()
        message(FATAL_ERROR "The generated Linux core must include its normal unity translation unit.")
    endif()

    add_executable(helengine_linux src/main.cpp src/linux_player.cpp src/linux_run_options.cpp src/linux_input_replay.cpp src/sdl_replay_gamepad.cpp src/linux_tracy_profiler.cpp)

    target_link_libraries(helengine_linux PRIVATE helengine_linux_generated_bridge helengine_generated_core)
    if(HELENGINE_NATIVE_PROFILE STREQUAL "Profiler")
        target_sources(helengine_linux PRIVATE src/linux_cpu_profile_sink.cpp)
