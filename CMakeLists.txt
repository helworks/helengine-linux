cmake_minimum_required(VERSION 3.25)
project(helengine_linux LANGUAGES NONE)

set(HELENGINE_CORE_CPP_ROOT "" CACHE PATH "Generated core root containing helcpp_config.hpp.")
set(HELENGINE_CODE_ROOT "" CACHE PATH "Generated project code-module root.")
set(HELENGINE_CONTENT_ROOT "" CACHE PATH "Cooked content root owned by the build pipeline.")
set(HELENGINE_LINUX_INITIAL_WIDTH "1280" CACHE STRING "Resolved initial desktop framebuffer width.")
set(HELENGINE_LINUX_INITIAL_HEIGHT "720" CACHE STRING "Resolved initial desktop framebuffer height.")
set(HELENGINE_NATIVE_PROFILE "Debug" CACHE STRING "Native profile: Debug, Release, or Profiler.")
set_property(CACHE HELENGINE_NATIVE_PROFILE PROPERTY STRINGS Debug Release Profiler)
option(HELENGINE_LINUX_BUILD_PLAYER "Build the real generated-core player, never a diagnostic substitute." ON)
if(NOT HELENGINE_NATIVE_PROFILE MATCHES "^(Debug|Release|Profiler)$")
    message(FATAL_ERROR "HELENGINE_NATIVE_PROFILE must be Debug, Release, or Profiler.")
endif()
if(HELENGINE_LINUX_BUILD_PLAYER)
    if(NOT HELENGINE_LINUX_INITIAL_WIDTH MATCHES "^[1-9][0-9]*$" OR NOT HELENGINE_LINUX_INITIAL_HEIGHT MATCHES "^[1-9][0-9]*$")
        message(FATAL_ERROR "Linux initial framebuffer dimensions must be positive integers.")
    endif()
    if(NOT EXISTS "${HELENGINE_CORE_CPP_ROOT}/helcpp_config.hpp")
        message(FATAL_ERROR "HELENGINE_CORE_CPP_ROOT must contain helcpp_config.hpp. Generate the Linux core first; diagnostics are available with HELENGINE_LINUX_BUILD_PLAYER=OFF.")
    endif()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The native Linux backend must compile on Linux from the real checkout.")
endif()
enable_language(CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(HELENGINE_NATIVE_PROFILE STREQUAL "Debug")
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "C++ build mode" FORCE)
else()
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "C++ build mode" FORCE)
endif()
find_package(Vulkan 1.1 REQUIRED)
find_package(SDL3 3.4.2 EXACT CONFIG REQUIRED)
find_program(HELENGINE_GLSLANG_VALIDATOR glslangValidator REQUIRED)
add_library(helengine_linux_graphics
    src/vulkan_context.cpp src/vulkan_renderer_2d.cpp src/sdl_host.cpp)
target_include_directories(helengine_linux_graphics PUBLIC src)
target_link_libraries(helengine_linux_graphics PUBLIC Vulkan::Vulkan SDL3::SDL3)
target_compile_options(helengine_linux_graphics PRIVATE -Wall -Wextra -Wpedantic -Werror=return-type)
target_compile_definitions(helengine_linux_graphics PUBLIC HELENGINE_LINUX_SHADER_ROOT="${CMAKE_CURRENT_BINARY_DIR}/shaders")
foreach(shader quad.vert quad.frag)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.spv"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
        COMMAND "${HELENGINE_GLSLANG_VALIDATOR}" -V --target-env vulkan1.1 "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}" -o "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.spv"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}" VERBATIM)
    list(APPEND compiled_shaders "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.spv")
endforeach()
add_custom_target(helengine_linux_shaders DEPENDS ${compiled_shaders})
add_executable(helengine_linux_gpu_test tests/gpu_test.cpp)
target_link_libraries(helengine_linux_gpu_test PRIVATE helengine_linux_graphics)
add_dependencies(helengine_linux_gpu_test helengine_linux_shaders)
add_executable(helengine_linux_sdl_test tests/sdl_host_test.cpp)
target_link_libraries(helengine_linux_sdl_test PRIVATE helengine_linux_graphics)
add_dependencies(helengine_linux_sdl_test helengine_linux_shaders)
enable_testing()
add_executable(helengine_linux_queue_test tests/queue_selection_test.cpp)
target_link_libraries(helengine_linux_queue_test PRIVATE Vulkan::Vulkan)
target_include_directories(helengine_linux_queue_test PRIVATE src)
add_test(NAME vulkan_queue_selection COMMAND helengine_linux_queue_test)
add_test(NAME vulkan_gpu_readback COMMAND helengine_linux_gpu_test)
if(EXISTS "${HELENGINE_CORE_CPP_ROOT}/RenderCommandList2D.hpp")
    add_library(helengine_linux_generated_bridge STATIC src/linux_render_manager_2d.cpp)
    target_include_directories(helengine_linux_generated_bridge PUBLIC "${HELENGINE_CORE_CPP_ROOT}" src)
    target_link_libraries(helengine_linux_generated_bridge PUBLIC helengine_linux_graphics)
endif()
if(HELENGINE_LINUX_BUILD_PLAYER)
    if(NOT EXISTS "${HELENGINE_CORE_CPP_ROOT}/KeyboardState.hpp")
        message(FATAL_ERROR "The generated Linux core must include DESKTOP_PLATFORM keyboard input. Regenerate using the Linux RuntimeGenerationContract.")
    endif()
    if(HELENGINE_NATIVE_PROFILE STREQUAL "Profiler")
        message(FATAL_ERROR "The Linux Profiler player requires Tracy integration, which is not complete; an uninstrumented player is not a Profiler build.")
    endif()
    if(EXISTS "${HELENGINE_CORE_CPP_ROOT}/helengine_core_unity.cpp")
        set(core_unity "${HELENGINE_CORE_CPP_ROOT}/helengine_core_unity.cpp")
    elseif(EXISTS "${HELENGINE_CORE_CPP_ROOT}/generated_unity.cpp")
        set(core_unity "${HELENGINE_CORE_CPP_ROOT}/generated_unity.cpp")
    else()
        message(FATAL_ERROR "The generated Linux core must include its normal unity translation unit.")
    endif()
    find_package(Threads REQUIRED)
    file(GLOB runtime_manifests CONFIGURE_DEPENDS "${HELENGINE_CORE_CPP_ROOT}/runtime/runtime_*_manifest.cpp")
    add_library(helengine_generated_core STATIC "${core_unity}" ${runtime_manifests})
    target_include_directories(helengine_generated_core PUBLIC "${HELENGINE_CORE_CPP_ROOT}")
    target_link_libraries(helengine_generated_core PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
    target_sources(helengine_linux_generated_bridge PRIVATE src/linux_render_manager_3d.cpp src/sdl_input_backend.cpp)
    add_executable(helengine_linux src/main.cpp src/linux_player.cpp src/linux_run_options.cpp)
    target_compile_definitions(helengine_linux PRIVATE HELENGINE_LINUX_INITIAL_WIDTH=${HELENGINE_LINUX_INITIAL_WIDTH} HELENGINE_LINUX_INITIAL_HEIGHT=${HELENGINE_LINUX_INITIAL_HEIGHT})
    target_link_libraries(helengine_linux PRIVATE helengine_linux_generated_bridge helengine_generated_core)
    add_dependencies(helengine_linux helengine_linux_shaders)
    add_executable(helengine_linux_input_test tests/input_test.cpp)
    target_link_libraries(helengine_linux_input_test PRIVATE helengine_linux_generated_bridge helengine_generated_core)
    add_test(NAME generated_sdl_input COMMAND helengine_linux_input_test)
    set_target_properties(helengine_linux PROPERTIES BUILD_RPATH "$ORIGIN/lib" INSTALL_RPATH "$ORIGIN/lib")
endif()
